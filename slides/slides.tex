\begin{frame}{因子工程：六大家族 (105+50+45+29+67+74 = 370 → KS drop 11 → \hl{359 维})}
  \vspace{2pt}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
    \vspace{6pt}
    \tiny
    \begin{itemize}\setlength{\itemsep}{2pt}\setlength{\topsep}{0pt}\setlength{\parsep}{0pt}
      \item \hl{F1 LOB 派生} (94 raw + 11 派生 = 105 维, \rhl{gain 40.4\%})\\
        OHLC, 10 档 bid/ask 价+挂单量, agg (avgbid/avgask/total*size), 4 mean, \texttt{midprice$_k$}, \texttt{spread$_k$}, \textbf{WMP 微价格} (Stoikov 2014): $\frac{a\cdot bs + b\cdot as}{bs+as}$, \texttt{bid\_diff/ask\_diff} 档内梯度 ── \emph{snapshot 表示, F2--F6 衍生底座}
      \item \hl{F2 多尺度 OFI} (0 raw + 50 派生, gain 5.5\%)\\
        \textbf{MLOFI} (Cont 2014, 多档 10$\times$3 $W$): $e_t^{(k)} = \mathbf{1}_{b\uparrow}\,bs - \mathbf{1}_{b\downarrow}\,bs^{-} + \cdots$, 窗口求和, $W\in\{5,20,60\}$ \\
        \textbf{EWMA-OFI} 多 $\alpha\in\{0.05,0.1,0.3,0.5\}$\,;\,\textbf{Kyle} $\lambda$ rolling \& \textbf{反 Kyle}\,$\frac{\Delta amt}{\sqrt[3]{|\Delta amt|\sigma}+\varepsilon}$
      \item \hl{F3 订单流强度} (18 raw + 27 派生, \rhl{gain 35.1\%})\\
        6 类事件 \texttt{l/m/c b/a} (limit/market/cancel × buy/sell) × \texttt{intst/ind/acc} 18 原始 + EWMA-intst 24 + cancel pressure 3 ── \emph{撮合层多空压力,\,\rhl{Top-4 全在此族}: \texttt{ma\_intst}=0.945, \texttt{mb\_intst}=0.860, \texttt{la/lb\_intst}=0.6}
      \item \hl{F4 微结构波动} (2 raw + 27 派生, gain 7.5\%)\\
        \texttt{volume/amount\_delta} (sign-log1p 重尾压制), signed RV $\sum r^2$, bipower $\sum|r_i||r_{i+1}|$, \textbf{Roll spread} $2\sqrt{-\text{cov}(r_t, r_{t-1})}$, vol-burst (短窗口/长窗口 vol 比) ── \emph{波动放大$\to$方向转折}
      \item \hl{F5 窗口统计} (0 raw + 67 派生, gain 5.2\%)\\
        100-tick \textbf{z-score} $(x-\mu)/\sigma$, \textbf{dual-z} (两窗口比), \textbf{quantile rank} 窗口内分位, sym-agnostic ── \emph{单族 gain 低但与所有族 $|\rho|{\le}0.07$,\,\rhl{OOD 稳健性 anchor}}
      \item \hl{F6 不对称性} (40 raw + 34 派生, gain 6.3\%)\\
        bid/ask rate L1--L10 (报价更新率), bsize/asize rate (挂单变化率), \textbf{GOFI} (Generalized OFI 含 cancel), EWMA residual (买卖侧不对称残差) ── \emph{买卖侧异质性}
    \end{itemize}

\begin{frame}{三类关键预处理 —— window-$z$ 单 trick 拉高 +22.88}
  \begin{columns}[T,onlytextwidth]
    \column{0.33\textwidth}
    \begin{block}{\hl{window-$z$ score}}
      \scriptsize
      所有统计量在 \textbf{100-tick 切片内}自适应归一化\\[2pt]
      \,\,$\Rightarrow$ 跨 stock 分布差异被窗口吸收\\[2pt]
      \,\,$\Rightarrow$ 训练外股票也能 zero-shot
      \begin{center}
        \vspace{2pt}
        \rhl{$+22.88$ PnL}
      \end{center}
    \end{block}
    \column{0.33\textwidth}
    \begin{block}{\hl{买卖镜像数据增强}}
      \scriptsize
      训练样本与 bid$\leftrightarrow$ask 互换 $+$ 标签反转的镜像拼接\\[2pt]
      \,\,$\Rightarrow$ 训练集 $\times 2$\\[2pt]
      \,\,$\Rightarrow$ 强制方向\textbf{对称先验}
      \begin{center}
        \vspace{6pt}
        \rhl{$+2.41$ PnL}
      \end{center}
    \end{block}
    \column{0.33\textwidth}
    \begin{block}{\hl{稳健归一化 (sign-log1p)}}
      \scriptsize
      \texttt{amount\_delta} 量级 $10^3$--$10^6$\,，重尾压制：
      \[
        v\mapsto\operatorname{sign}(v)\log(1{+}|v|)
      \]
      \,\,$\Rightarrow$ NN/LGB 梯度数值稳定\\[3pt]
      \,\,$\Rightarrow$ LGB 大幅受益, NN 中性
    \end{block}
  \end{columns}
  \vspace{6pt}
  \begin{center}
    \footnotesize\itshape
    主题：\textbf{在输入侧让模型``看不见''非平稳分布偏移}\,—— \hl{范化性}的根本。
  \end{center}
\end{frame}

\begin{frame}{模型工程 (A)：NN 架构 —— 纯 MLP 的设计启示}
  \vspace{-4pt}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
    \scriptsize
    \textbf{\color{icmlblue}本任务的 4 个客观属性}
    \begin{enumerate}\setlength{\itemsep}{1pt}
      \item 数据集小（5 stock $\times$ 120 day）
      \item 训练/测试股票可能切换，\textbf{难利用跨股票信息}
      \item 评测时\textbf{只能用 100 ticks 窗口信息}
      \item 因子已充分蒸馏潜在信息，\textbf{MLP 非线性已足够}
    \end{enumerate}

    \column{0.46\textwidth}
    \scriptsize
    \textbf{\color{icmlblue}从而推断}
    \begin{enumerate}\setlength{\itemsep}{1pt}
      \item 复杂\hl{时序}架构 (CNN/RNN/TCN) 用处小（只 100 ticks，因子已聚合）
      \item 复杂\hl{跨通道}架构 (Transformer) 发挥空间小（跨 stock 信息难用）
      \item 复杂架构容易\rhl{过拟合}小数据
    \end{enumerate}
  \end{columns}
  \vspace{-2pt}
  \begin{center}\footnotesize
    $\Rightarrow$ 只用 4-layer MLP；实验也表明换更复杂架构都\textbf{没有更好}。

\begin{frame}{训练核心：SPO+ 决策焦点学习 —— Phase 1 L2 预训练 $\to$ Phase 2 SPO+ finetune}
  \vspace{-6pt}
  \begin{columns}[T,onlytextwidth]

    \scriptsize
    \textbf{\color{icmlblue}两阶段训练流程}
    \begin{enumerate}\setlength{\itemsep}{2pt}
      \item \hl{Phase 1 —— L2 回归预训练}\\
        \scriptsize 目标 $y=(\mathrm{mp}_{t+60}-\mathrm{mp}_t)/(\mathrm{mp}_t+1)$
      \item \hl{Phase 2 —— SPO+ joint finetune}\\
        \scriptsize warm-start,\,\textbf{全量数据}(train+val,\,0--119) SPO+ 联合优化
    \end{enumerate}

    \vspace{2pt}
    \textbf{\color{icmlblue}SPO+ surrogate (Elmachtoub \& Grigas 2022)：}
    \[
      \ell_{\mathrm{SPO+}} = \mathrm{ReLU}\!\big(|2\hat y{-}y|{-}f\big) - z^{\!\star}(y)(2\hat y{-}y) + f|z^{\!\star}(y)|
    \]

    \vspace{1pt}
    \textbf{\color{icmlblue}两个好处}
    \begin{itemize}\setlength{\itemsep}{1pt}
      \item \hl{(1) 对齐买卖 PnL}
      \item \hl{(2) 全数据 Data scaling}
    \end{itemize}

    \begin{center}
      \rhl{$+3.54$ PnL}
    \end{center}
  \end{columns}

      \[
      \hat a =
      \begin{cases}
        +1\,(\text{买}), & \hat y > +\theta_{\mathrm{up}}\\

        \phantom{+}0, & \text{otherwise}
      \end{cases}
      \]
      \vspace{-2pt}
      \hl{$\theta_{\mathrm{up}}{=}3.00{\times}10^{-4}$}\,,\,\,\hl{$\theta_{\mathrm{dn}}{=}2.16{\times}10^{-4}$}
    \end{block}
